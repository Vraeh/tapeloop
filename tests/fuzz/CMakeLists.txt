function(tapeloop_add_fuzzer name)
  add_executable(${name})
  target_sources(${name} PRIVATE ${ARGN} FuzzInput.hpp)
  target_link_libraries(${name} PRIVATE tapeloop-core)
  target_link_options(${name} PRIVATE -fsanitize=fuzzer)
  tapeloop_core_warnings(${name})
endfunction()

# The readers of replay files, through to the FFmpeg the plugin decodes with.
function(tapeloop_add_ffmpeg_fuzzer name)
  tapeloop_add_fuzzer(${name} ${ARGN} ReplayDamage.hpp StoredReplay.hpp)
  target_link_libraries(${name} PRIVATE tapeloop-decode FFmpeg::avutil)
endfunction()

tapeloop_add_fuzzer(replay-format-fuzz ReplayFormatFuzz.cpp ReplayDamage.hpp)
